만든 사이트를 스스로 검수하는 법

저희가 파는 팩을 재는 잣대 그대로입니다. 붙여 넣기 한 번이면 됩니다.
스펙팩(07_AI빌드_스펙팩.md)을 안 여셔도 이 파일 하나면 됩니다.

무엇을 재나 — 열두 가지

재는 것어떻게
위 메뉴가 굴려도 따라오나position 이 sticky·fixed 인지 봅니다
본문과 푸터가 맞붙었나마지막 알맹이와 푸터 사이를 픽셀로 잽니다
덩어리 사이 간격이 어긋났나그 칸의 «리듬»을 재고 벗어난 것만 짚습니다
버튼·배지가 늘어났나글 폭과 실제 폭을 견줍니다. 동그란 버튼이 타원이 됐는지도
좌우로 미는 칸에 막대가 드러났나화살표가 있는데 띠까지 깔렸는지 봅니다
표가 제 칸을 넘쳤나표 폭과 담는 칸 폭을 견줍니다
지금 보는 메뉴가 켜져 있나클래스 이름이 아니라 «옆 링크와 달라 보이는지»로 봅니다
배경 위 글자가 읽히나WCAG 대비 — 본문 4.5, 큰 글자 3.0
나란한 사진 크기가 같은가윗변이 같은 것끼리만 높이를 견줍니다. 사진이 아직 안 들어간 «자리표시자»도 셉니다
한 화면 안에서 콘텐츠 폭이 갈리나본문 덩어리들의 왼쪽 끝이 몇 번 움직이는지 셉니다. 셋부터 짚습니다
나란히 선 칸들의 제목 높이가 맞나한 줄에 선 것끼리만 견줍니다 — 한 칸에만 배지가 붙으면 그 칸만 밀립니다
끊어진 링크·없는 그림·외톨이 화면파일만 봐도 아는 것이라 한 번에 전부 잽니다
우리 팩으로 만든 것이 아니어도 됩니다. 클래스 이름을 보지 않고 «생김새»로만 재기 때문에, 예전에 만드신 사이트에도 그대로 돌아갑니다.

0먼저 — 만든 화면을 브라우저로 엽니다

HTML 파일이면 index.html 을 더블클릭하시면 됩니다. Next·React 처럼 서버가 필요한 것이면 터미널에 npm run dev 를 치고 localhost:3000 을 여세요.

그 화면에서 F12 를 누르면 창이 갈라집니다. 위 탭에서 Console 을 고르면 > 가 깜빡이는 빈 줄이 나옵니다. 거기에 붙여 넣습니다.

// node 로 돌리세요:  node 검수.mjs ./내사이트
// ⚠ «사이트 뿌리»를 주세요. 하위 폴더까지 훑습니다 —
//    pages/ 만 주면 한 층 위의 화면 목록(index.html)을 못 봐서
//    거기서만 이어지는 화면이 죄다 «외톨이»로 잘못 잡힙니다.
import { readdirSync, readFileSync, existsSync, statSync } from "node:fs";
import { join, resolve, dirname, relative } from "node:path";

const 뿌리 = process.argv[2] || ".";
const 쪽들 = [];
(function 훑기(방) {
  for (const e of readdirSync(방, { withFileTypes: true })) {
    if (e.name.startsWith(".") || e.name === "node_modules") continue;
    const 길 = join(방, e.name);
    if (e.isDirectory()) 훑기(길);
    else if (e.name.endsWith(".html")) 쪽들.push(길);
  }
})(뿌리);

const 흠 = [];
const 이어진곳 = new Set();

for (const 길 of 쪽들) {
  const 쪽 = relative(뿌리, 길);
  const s = readFileSync(길, "utf8");

  // ① 끊어진 링크
  for (const m of s.matchAll(/href="([^"]+)"/g)) {
    const v = m[1];
    if (!v || v.startsWith("#") || v.startsWith("http") || v.startsWith("mailto:")) continue;
    const 갈곳 = resolve(dirname(길), v.split("#")[0].split("?")[0]);
    이어진곳.add(갈곳);
    if (v.endsWith(".html") && !existsSync(갈곳)) 흠.push("[끊어진 링크] " + 쪽 + " -> " + v);
  }

  // ② 없는 그림
  for (const m of s.matchAll(/src="([^"]+)"/g)) {
    const v = m[1];
    if (!v || v.startsWith("http") || v.startsWith("data:")) continue;
    if (!existsSync(resolve(dirname(길), v.split("?")[0]))) 흠.push("[없는 그림] " + 쪽 + " -> " + v);
  }

  // ③ 눌러도 아무 데도 안 가는 것
  const 빈링크 = (s.match(/href="(#|javascript:void\(0\))"/g) || []).length;
  if (빈링크) 흠.push("[빈 링크] " + 쪽 + " 에 " + 빈링크 + "개 — 갈 곳이 있으면 잇고, 없으면 «안 눌리게» 두세요");

  // ④ 0바이트
  if (statSync(길).size === 0) 흠.push("[빈 파일] " + 쪽);
}

// ⑤ 외톨이 화면 — 만들어 놓고 아무 데서도 안 이어지는 쪽
for (const 길 of 쪽들) {
  const 쪽 = relative(뿌리, 길);
  if (/^(index|home)\./i.test(쪽)) continue;
  if (!이어진곳.has(resolve(길))) 흠.push("[외톨이] " + 쪽 + " 은 어느 화면에서도 이어지지 않습니다");
}

console.log("화면 " + 쪽들.length + "장 · 흠 " + 흠.length + "건");
for (const h of 흠) console.log("  " + h);
if (!흠.length) console.log("  깨끗합니다");
